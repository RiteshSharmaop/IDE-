import axios from "axios";

const BACKEND_URL = import.meta.env.VITE_BACKEND_URL || "http://localhost:8080";

const API = axios.create({
  baseURL: BACKEND_URL,
});

// Add auth interceptor
API.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem("token");
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

const LANGUAGE_VERSIONS = {
  javascript: "18.15.0",
  typescript: "5.0.3",
  python: "3.10.0",
  java: "15.0.2",
  csharp: "6.12.0",
  cpp: "10.2.0",
};

export const runTheCode = async (language, sourceCode, input = "") => {
  try {
    const res = await API.post("/execute/run", {
      code: sourceCode,
      language,
      input,
    });

    if (!res.data.success) {
      throw new Error(res.data.message || "Code execution failed");
    }

    return res.data.data;
  } catch (error) {
    // Better error handling
    let errorMessage = error.message;
    
    if (error.response?.status === 400) {
      errorMessage = error.response.data.message || "Invalid request";
    } else if (error.response?.status === 429) {
      errorMessage = "Too many requests. Please wait before running again.";
    } else if (error.response?.status === 503) {
      errorMessage = "Code execution service is temporarily unavailable.";
    } else if (error.response?.data?.message) {
      errorMessage = error.response.data.message;
    }

    console.error("❌ Error running code:", errorMessage);
    throw new Error(errorMessage);
  }
};






// pistone code
import axios from "axios";

const API = axios.create({
  baseURL: "https://emkc.org/api/v2/piston",
});

const LANGUAGE_VERSIONS = {
  javascript: "18.15.0",
  typescript: "5.0.3",
  python: "3.10.0",
  java: "15.0.2",
  csharp: "6.12.0",
  cpp: "10.2.0",
};

export const runTheCode = async (language, sourceCode, input = "") => {
  try {
    console.log("🚀 Running code:", { language, codeLength: sourceCode.length, hasInput: !!input });
    const res = await API.post("/execute", {
      language,
      version: LANGUAGE_VERSIONS[language],
      files: [{ content: sourceCode }],
      stdin: input, // 👈 this is where you send user input
    });
    console.log("✅ Code execution response:", res.data);
    return res.data.run;
  } catch (error) {
    console.error("❌ Error running code:", error.response?.data || error.message);
    throw error;
  }
};
